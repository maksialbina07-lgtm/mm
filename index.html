#!/usr/bin/env python3 """ publish_apk.py — делает страницу index.html для скачивания APK. Примеры: python publish_apk.py app-debug.apk python publish_apk.py app-debug.apk --out site --title "Моё приложение" --version 1.0 python publish_apk.py a.apk b.apk --base-url https://example.com/app python publish_apk.py app-debug.apk --serve 8080 # сразу раздать по http://:8080 Результат: папка (по умолчанию ./site) с index.html и копиями APK. Её целиком можно залить на любой хостинг / nginx / GitHub Pages. """ import argparse import datetime import hashlib import html import http.server import shutil import socketserver import sys import urllib.parse import zipfile from functools import partial from pathlib import Path PAGE = """ {title}

{title}

{subtitle}

{cards}

Как установить

  1. Скачай APK на телефон с Android.
  2. Открой файл. Если система попросит, разреши установку из этого источника.
  3. Подтверди установку.
""" CARD = """
{name}
{meta}
⬇ Скачать APK
SHA-256{sha}
""" def human_size(n: int) -> str: for unit in ("Б", "КБ", "МБ", "ГБ"): if n < 1024 or unit == "ГБ": return f"{n:.0f} {unit}" if unit == "Б" else f"{n:.1f} {unit}" n /= 1024 def sha256(path: Path) -> str: h = hashlib.sha256() with open(path, "rb") as f: for chunk in iter(lambda: f.read(1 << 20), b""): h.update(chunk) return h.hexdigest() def check_apk(path: Path): if not path.is_file(): sys.exit(f"Файл не найден: {path}") if not zipfile.is_zipfile(path) or "AndroidManifest.xml" not in zipfile.ZipFile(path).namelist(): sys.exit(f"Это не похоже на APK: {path}") def build(apks, out: Path, title, subtitle, version, base_url): out.mkdir(parents=True, exist_ok=True) base = base_url.rstrip("/") + "/" if base_url else "" cards, today = [], datetime.date.today().strftime("%d.%m.%Y") for apk in apks: check_apk(apk) dst = out / apk.name if apk.resolve() != dst.resolve(): shutil.copy2(apk, dst) meta = [human_size(dst.stat().st_size), today] if version: meta.insert(0, f"версия {version}") cards.append(CARD.format( name=html.escape(apk.name), meta=html.escape(" · ".join(meta)), href=html.escape(base + urllib.parse.quote(apk.name), quote=True), sha=sha256(dst))) page = PAGE.format(title=html.escape(title), subtitle=html.escape(subtitle), cards="\n".join(cards)) (out / "index.html").write_text(page, encoding="utf-8") return out / "index.html" class Handler(http.server.SimpleHTTPRequestHandler): extensions_map = {**http.server.SimpleHTTPRequestHandler.extensions_map, ".apk": "application/vnd.android.package-archive"} def serve(directory: Path, port: int): with socketserver.ThreadingTCPServer(("0.0.0.0", port), partial(Handler, directory=str(directory))) as httpd: print(f"🌐 Раздаю {directory} на порту {port}: http://localhost:{port}/ (Ctrl+C — стоп)") try: httpd.serve_forever() except KeyboardInterrupt: print("\nОстановлено.") def main(): ap = argparse.ArgumentParser(description="Страница index.html для скачивания APK") ap.add_argument("apk", nargs="+", type=Path, help="один или несколько .apk") ap.add_argument("--out", type=Path, default=Path("site"), help="папка результата (по умолчанию ./site)") ap.add_argument("--title", default="Скачать приложение") ap.add_argument("--subtitle", default="Android-приложение (APK)") ap.add_argument("--version", default="", help="версия для подписи, например 1.0") ap.add_argument("--base-url", default="", help="адрес папки на сервере, например https://example.com/app " "(если пусто — ссылки относительные)") ap.add_argument("--serve", type=int, metavar="PORT", help="сразу раздать папку по HTTP на этом порту") args = ap.parse_args() index = build(args.apk, args.out, args.title, args.subtitle, args.version, args.base_url) print(f"✅ Готово: {index}") if args.base_url: print(f"🔗 Страница будет по адресу: {args.base_url.rstrip('/')}/") if args.serve: serve(args.out, args.serve) if __name__ == "__main__": main()
